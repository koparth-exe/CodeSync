## 1. Core Normalized Submission

The canonical, platform-agnostic data structure generated by the normalization pipeline:

```typescript
export interface NormalizedSubmission {
  /** Unique submission tracking ID (UUID v4) */
  readonly id: string;

  // ─── Platform Context ───
  readonly platform: PlatformId;
  readonly submissionId: string;
  readonly submissionUrl: string;

  // ─── Target GitHub Configuration (Snapshotted at detection) ───
  readonly targetRepository: string; // e.g. "octocat/dsa-solutions"
  readonly targetBranch: string;     // e.g. "main"

  // ─── Problem Metadata ───
  readonly problemTitle: string;     // Sanitized human-readable title
  readonly problemSlug: string;      // Canonical URL-safe slug (e.g. "two-sum")
  readonly problemId: string;        // Platform ID (e.g. "1", "4A")

  // ─── Verdict & Code ───
  readonly status: SubmissionStatus;
  readonly language: Language;
  readonly sourceCode: string;       // UTF-8 sanitized source code
  readonly contentHash: string;      // SHA-256 hash of source code (LF normalized)

  // ─── Extraction Provenance ───
  readonly extractionMethod: ExtractionMethod;
  readonly extractionConfidence: number; // 0.00 to 1.00

  // ─── Optional Metadata ───
  readonly difficulty?: string;      // "Easy", "Medium", "Hard"
  readonly rating?: number;          // e.g. Codeforces 800 - 3500
  readonly contestId?: string;
  readonly runtime?: string;
  readonly memory?: string;
  readonly tags?: readonly string[];

  // ─── Timestamps ───
  readonly submittedAt: number;      // Unix ms on platform
  readonly detectedAt: number;       // Unix ms when extension captured event

  // ─── Platform-Specific Extra ───
  readonly platformMetadata?: Readonly<Record<string, unknown>>;
}
```

## 2. Queue & Hybrid Storage Partitioning Models

### 2.1 Queue Item Metadata (`browser.storage.local`)

Stores lightweight state indices in `browser.storage.local` under key `codesync:queue:metadata`:

```typescript
export interface QueueItemMetadata {
  readonly id: string;               // UUID v4
  readonly payloadId: string;        // Reference to IndexedDB payload ("payload:<id>")
  readonly platform: PlatformId;
  readonly problemSlug: string;
  readonly problemTitle: string;
  readonly targetRepository: string;
  readonly targetBranch: string;
  readonly language: Language;
  readonly status: SubmissionStatus;
  readonly contentHash: string;      // SHA-256 for rapid duplicate checks

  // State Management
  state: QueueState;
  attempts: number;
  crashCount: number;                // Track crashes to quarantine poison pills
  lastError?: CodeSyncError;
  nextRetryAt?: number;              // Timestamp ms for backoff

  // Results
  commitSha?: string;
  commitUrl?: string;

  // Timestamps
  readonly createdAt: number;
  updatedAt: number;
  completedAt?: number;
}

export enum QueueState {
  PENDING = 'pending',
  PROCESSING = 'processing',
  COMPLETED = 'completed',
  FAILED = 'failed',
  REQUIRES_ATTENTION = 'requires_attention',
  SKIPPED = 'skipped',
}
```

### 2.2 Queue Item Payload (`IndexedDB`)

Stores heavy source code payloads in the `codesync_db` IndexedDB under the `payloads` object store:

```typescript
export interface QueueItemPayload {
  readonly id: string;               // Key: "payload:<uuid>"
  readonly sourceCode: string;       // Full source code string (up to 500 KB)
  readonly platformMetadata?: Record<string, unknown>;
  readonly createdAt: number;
}
```

### 2.3 Persistent Queue Lease Record (`browser.storage.local`)

Stored under `codesync:queue:lease`:

```typescript
export interface QueueLeaseRecord {
  readonly workerId: string;          // Unique UUID v4 for the worker instance
  readonly acquiredAt: number;        // Timestamp ms of acquisition
  expiresAt: number;                  // Heartbeat expiry (acquiredAt + LEASE_TTL_MS)
  readonly triggerSource: 'alarm' | 'submission_event' | 'manual_retry' | 'startup';
}
```

## 4. Extension Messaging Schemas

Strict message typing across extension boundaries:

```typescript
export type ExtensionMessageType =
  // Content Script -> Background
  | 'SUBMISSION_DETECTED'
  | 'EXTRACTION_COMPLETED'
  | 'EXTRACTION_FAILED'

  // Background -> Content Script
  | 'REQUEST_RECOVERY_SELECTION'

  // UI -> Background
  | 'GET_STATUS'
  | 'GET_QUEUE_METADATA'
  | 'GET_HISTORY'
  | 'RETRY_QUEUE_ITEM'
  | 'CANCEL_QUEUE_ITEM'
  | 'PURGE_COMPLETED'
  | 'UPDATE_CONFIG'
  | 'START_DEVICE_AUTH'
  | 'POLL_DEVICE_AUTH'
  | 'LOGOUT'

  // Background -> UI
  | 'STATUS_CHANGED'
  | 'QUEUE_UPDATED'
  | 'AUTH_STATE_CHANGED';

export interface ExtensionMessage<T = unknown> {
  readonly id: string;               // UUID v4 nonce
  readonly type: ExtensionMessageType;
  readonly payload: T;
  readonly timestamp: number;        // Replay validation (<30s window)
  readonly senderContext: 'content-script' | 'popup' | 'options' | 'background';
}
```

## 5. Error Taxonomy & Fail-Closed Models

```typescript
export interface CodeSyncError {
  readonly code: ErrorCode;
  readonly message: string;
  readonly category: ErrorCategory;
  readonly retryable: boolean;
  readonly failClosed: boolean;      // True if operation must halt rather than guess
  readonly suggestedActions: readonly RecoveryAction[];
  readonly context?: Record<string, string | number | boolean>; // NEVER contain secrets/code
```

## 6. Duplicate Detection Policies

```typescript
export enum DuplicatePolicy {
  /** Skip commit if remote content SHA-256 is identical. Update if different. */
  REPLACE_IF_DIFFERENT = 'replace_if_different',

  /** Always create a commit even if content is identical. */
  ALWAYS_REPLACE = 'always_replace',

  /** Keep all submissions by appending sequential version suffixes (e.g. -v2.cpp). */
  KEEP_ALL = 'keep_all',

  /** Never overwrite existing files. Skip if file exists regardless of content. */
  CREATE_ONLY = 'create_only',
}
```
